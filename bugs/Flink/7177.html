Title

batch query cannot generate plan when a sorted view into multi sinks

Description
A batch query  write a sorted view into multi sinks will get a cannot plan exception 

  
 
  @Test
  def testSortedResultIntoMultiSinks(): Unit = {
    util.tableEnv.executeSql(
      s"""
         |CREATE TABLE Src (
         |  `a` INT,
         |  `b` BIGINT,
         |  `c` STRING,
         |  `d` STRING,
         |  `e` STRING
         |) WITH (
         |   'connector'  =  'values' ,
         |   'bounded'  =  ' true ' 
         |)
       """.stripMargin)
    val query =  "SELECT * FROM Src order by c" 
    val table = util.tableEnv.sqlQuery(query)
    util.tableEnv.registerTable( "sortedTable" , table)
    util.tableEnv.executeSql(
      s"""
         |CREATE TABLE sink1 (
         |  `a` INT,
         |  `b` BIGINT,
         |  `c` STRING
         |) WITH (
         |   'connector'  =  'filesystem' ,
         |   'format'  =  'testcsv' ,
         |   'path'  =  '/tmp/test' 
         |)
       """.stripMargin)
    util.tableEnv.executeSql(
      s"""
         |CREATE TABLE sink2 (
         |  `a` INT,
         |  `b` BIGINT,
         |  `c` STRING,
         |  `d` STRING
         |) WITH (
         |   'connector'  =  'filesystem' ,
         |   'format'  =  'testcsv' ,
         |   'path'  =  '/tmp/test' 
         |)
      """.stripMargin)
    val stmtSet= util.tableEnv.createStatementSet()
    stmtSet.addInsertSql(
       "insert into sink1 select a, b, listagg(d) from sortedTable group by a, b" )
    stmtSet.addInsertSql(
       "insert into sink2 select a, b, c, d from sortedTable" )

    util.verifyExecPlan(stmtSet)
  }
