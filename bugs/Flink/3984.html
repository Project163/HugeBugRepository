Title

Validation error when insert decimal/varchar with precision into sink using TypeInformation of row

Description
Test code like follwing(in blink planner): 
  
 
		tEnv.sqlUpdate( "create table randomSource ("  +
						 "		a varchar(10),"  +
						 "		b decimal(20,2)"  +
						 "	) with ("  +
						 "		 'type'  =  'random' ,"  +
						 "		 'count'  =  '10' "  +
						 "	)" );
		tEnv.sqlUpdate( "create table printSink ("  +
						 "		a varchar(10),"  +
						 "		b decimal(22,2),"  +
						 "	) with ("  +
						 "	 'type'  =  'print' "  +
						 "	)" );
		tEnv.sqlUpdate( "insert into printSink select * from randomSource" );
		tEnv.execute("");
