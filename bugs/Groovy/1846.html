Title

CompileStatic gives VerifyError: Register 1 contains wrong type

Description

  import  groovy.transform.*

@CompileStatic
 class  Test {
  List< Integer > elements
  
   public  Test( List< Integer > elements ) {
     this .elements = elements
  }
  
  void regularEach() {
     int  idx = 0
    elements.each {  Integer  elem ->
       "$elem:${idx++}" 
    }
  }
}

 new  Test( [1,2,3] )
