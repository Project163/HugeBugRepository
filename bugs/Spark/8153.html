Title

SQL sequence function with interval returns unexpected error in latest versions

Description
For example this returns  java.lang.ArrayIndexOutOfBoundsException: 1   
  
 
select sequence(
 date_trunc( 'month' ,  '2021-08-30' ),
 date_trunc( 'month' ,  '2021-08-15' ),
 - interval 1 month) 
  
 Another cases like - all ok 
  
 
select sequence(
 date_trunc( 'month' ,  '2021-07-15' ),
 date_trunc( 'month' ,  '2021-08-30' ),
 interval 1 month) as x

 , sequence(
 date_trunc( 'month' ,  '2021-08-30' ),
 date_trunc( 'month' ,  '2021-07-15' ),
 - interval 1 month) as y

 , sequence(
 date_trunc( 'month' ,  '2021-08-15' ),
 date_trunc( 'month' ,  '2021-08-30' ),
 interval 1 month) as z
