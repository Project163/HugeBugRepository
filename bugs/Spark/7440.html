Title

Description
val  spark: SparkSession = SparkSession
      .builder()
      .master( "local" )
      .appName( "SparkByExamples.com" )
      .getOrCreate()
    spark.sparkContext.setLogLevel( "ERROR" )

     import  spark.sqlContext.implicits._

     val  df = Seq((-0.0, 0.0)).toDF( "neg" ,  "pos" )
