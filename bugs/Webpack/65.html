Title

Webpack2, import and shorthand properties

Description
Hello !

It seems there is a problem with shorthand properties :

```
js
import { combineReducers } from 'redux';
import projects from './projects';

const rootReducer = combineReducers({
... [Log Snipped] ...

export default rootReducer;
```

generates:

```
js
"use strict";
    /* harmony import */ var __WEBPACK_IMPORTED_MODULE_0_redux__ = __webpack_require__(/*! redux */ 172);
    /* harmony import */ var __WEBPACK_IMPORTED_MODULE_0_redux___default = __WEBPACK_IMPORTED_MODULE_0_redux__ && __WEBPACK_IMPORTED_MODULE_0_redux__.__esModule ? function() { return __WEBPACK_IMPORTED_MODULE_0_redux__['default'] } : function() { return __WEBPACK_IMPORTED_MODULE_0_redux__; }
    /* harmony import */ Object.defineProperty(__WEBPACK_IMPORTED_MODULE_0_redux___default, 'a', { get: __WEBPACK_IMPORTED_MODULE_0_redux___default });
... [Log Snipped] ...

    /* harmony default export */ exports["a"] = rootReducer;
```

The problem is here:

``` js
var rootReducer = /* harmony import */__WEBPACK_IMPORTED_MODULE_0_redux__["combineReducers"].bind()({
        /* harmony import */__WEBPACK_IMPORTED_MODULE_1__projects__["a"]
    })
```

The key `projets` is lost. Expecting:

``` js
var rootReducer = /* harmony import */__WEBPACK_IMPORTED_MODULE_0_redux__["combineReducers"].bind()({
        projects: /* harmony import */__WEBPACK_IMPORTED_MODULE_1__projects__["a"]
    })
```
