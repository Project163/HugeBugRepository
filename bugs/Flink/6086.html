Title

Lookup join + WITH clause throws NPE

Description
Add the following SQL to  org.apache.flink.table.api.TableEnvironmentITCase  to reproduce this bug. 
  
 
@Test
 def  myTest(): Unit = {
  tEnv.executeSql(
    """
      |CREATE TABLE T1 (
      |  a INT,
      |  b STRING,
      |  proctime AS PROCTIME()
      |) WITH (
      |  'connector' = 'values',
      |  'bounded' = ' true '
      |)
      |""".stripMargin)
  tEnv.executeSql(
    """
      |CREATE TABLE T2 (
      |  a INT,
      |  b STRING
      |) WITH (
      |  'connector' = 'values',
      |  'bounded' = ' true '
      |)
      |""".stripMargin)
  tEnv.explainSql(
    """
      |WITH MyView(a, b) AS (SELECT a, b FROM T2)
      |SELECT * FROM T1 AS T
      |LEFT JOIN MyView FOR SYSTEM_TIME AS OF T.proctime AS D
      |ON T.a = D.a
      |""".stripMargin)
}
