   <soapenv:Envelope xmlns:wsa="[URL] xmlns:soapenv="[URL]
      <soapenv:Header> 
         <wsa:MessageID>urn:uuid: </wsa:MessageID> 
         <wsa:Action>urn:getQuote</wsa:Action> 
      </soapenv:Header> 
      <soapenv:Body>

</soapenv:Envelope>
