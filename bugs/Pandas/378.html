Title

Time series dtype bug in DataFrame constructor

Description
from pandas import Series, DataFrame
import pandas as pd

rng1 = pd.date_range('1/1/1999', '1/1/2012', freq='MS')
s1 = Series(np.random.randn(len(rng1)), rng1)

rng2 = pd.date_range('1/1/1980', '12/1/2001', freq='MS')
s2 = Series(np.random.randn(len(rng2)), rng2)

df = DataFrame({'s1' : s1, 's2' : s2})
# s1 is fine

In [39]: s1.index.dtype
Out[39]: dtype('datetime64[ns]')

In [41]: s1.index.values.dtype
Out[41]: dtype('datetime64[ns]')
# s2 is fine

In [40]: s2.index.dtype
Out[40]: dtype('datetime64[ns]')

In [42]: s2.index.values.dtype
Out[42]: dtype('datetime64[ns]')
# df.index looks fine

In [43]: df.index.dtype
Out[43]: dtype('datetime64[ns]')
# WAT?!

In [44]: df.index.values.dtype
Out[44]: dtype('datetime64[us]')
