Title

Static type checker has problems with args coerced to arrays

Description
import  groovy.transform.*

@TypeChecked
def m() {
     try  {
         throw   new  Exception()
    }  catch  (Throwable t) {
        def newTrace = []
        def clean = newTrace.toArray(newTrace as StackTraceElement[])
    }
}

m()
