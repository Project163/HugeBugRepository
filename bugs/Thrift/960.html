Title

Description

 struct Element {
	1: string foobar
}

struct Test {
	1: list<Element>  elms
}

   Test.cs

{
   //... stuff ...

  {
     //... stuff ...
          case  1:

               for (  int  _i1 = 0; _i1 < _list0.Count; ++_i1)

 }
