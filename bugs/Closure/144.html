Title

Closure Compiler Advanced Mode does not recognize toJSON (like it recognizes toString)

Description
What steps will reproduce the problem? 
Compile the following simple javascript program (with advanced optimizations):

// ==ClosureCompiler==
// @compilation_level ADVANCED_OPTIMIZATIONS
// @output_file_name default.js
// ==/ClosureCompiler==

/**
@constructor
@param {number} x
@param {number} y
*/
function foo(x, y) { 
  this.x = x;
  this.y = y;
  this.lastSum = x + y;
};

/**
@type {number}
*/
foo.prototype.x;

/**
@type {number}
*/
foo.prototype.y;

/**
@type {number}
*/
foo.prototype.lastSum;

/**
@override
@return {string}
*/
foo.prototype.toString = function() {
  return '(' + this.x + ',' + this.y + ')'; 
};

/**
@return {Object}
*/
foo.prototype.toJSON = function() {
  return {
    x: this.x,
    y: this.y
  };
};

/**
@return {number}
*/
foo.prototype.recalc = function() {
  return (this.lastSum = this.x + this.y);
};

var temp = new foo(10, 12);
alert(temp.toString() + '\n\n' + JSON.stringify(temp));
temp.x = 15;
alert(temp.toString() + '\n\n' + JSON.stringify(temp));
temp.recalc();
alert(temp.toString() + '\n\n' + JSON.stringify(temp));

// *** EOF 

 What is the expected output? What do you see instead? 

The output produced (with pretty print) is:
function a(c, d) {
  this.x = c;
  this.y = d;
  this.a = c + d
}
a.prototype.toString = function() {
  return"(" + this.x + "," + this.y + ")"
};
a.prototype.recalc = function() {
  return this.a = this.x + this.y
};
var b = new a(10, 12);
alert(b.toString() + "\n\n" + JSON.stringify(b));
b.x = 15;
alert(b.toString() + "\n\n" + JSON.stringify(b));
b.recalc();
alert(b.toString() + "\n\n" + JSON.stringify(b));
