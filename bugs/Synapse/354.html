Title

Pattern attribute of the property mediator does not seem to work

Description
Steps to reproduce:  
================  
1. Create a sequence like the following  
<sequence xmlns="[URL] name="main">  
   <in>  
      <switch xmlns:ns="[URL] xmlns:ns3="[URL] xmlns:m0="[URL] source="//m0:getQuote/m0:request/m0:symbol">  
         <case regex="IBM">  
            <property name="symbol" value="Great stock - IBM" />  
         </case>  
         <case regex="MSFT">  
            <property name="symbol" value="Are you sure? - MSFT" />  
         </case>  
         <default>  
            <property name="symbol" expression="fn11:concat('Normal Stock - ', //m0:getQuote/m0:request/m0:symbol)" scope="default" />  
         </default>  
      </switch>  
      <property xmlns:ns="[URL] xmlns:ns3="[URL] xmlns:m0="[URL] name="MyProp" expression="//m0:getQuote/m0:request/m0:symbol" pattern=" [a-zA-Z_0-9] " scope="default" />  
      <log level="custom">  
         <property xmlns:ns="[URL] xmlns:ns3="[URL] name="symbol" expression="get-property('symbol')" />  
         <property xmlns:ns="[URL] xmlns:ns3="[URL] name="epr" expression="get-property('To')" />  
         <property xmlns:ns="[URL] xmlns:ns3="[URL] name="customProperty" expression="get-property('MyProp')" />  
      </log>  
      <send />  
   </in>  
   <out>  
      <send />  
   </out>  
</sequence>  

 2. Note the following property mediator in the above sequence. I used pattern attribute in there to evaluate the result of xpath expression against a regex  
<property xmlns:ns="[URL] xmlns:ns3="[URL] xmlns:m0="[URL] name="MyProp" expression="//m0:getQuote/m0:request/m0:symbol" pattern=" [a-zA-Z_0-9] " scope="default" />
