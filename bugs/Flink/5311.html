Title

PushFilterIntoTableSourceScanRule fails to deal with NULLs

Description
Add the following test case to  PushFilterIntoTableSourceScanRuleTest  to reproduce this bug: 

  
 
@Test
 public  void myTest() {
     String  ddl =
             "CREATE TABLE MTable (" 
                    +  "  a STRING," 
                    +  "  b STRING" 
                    +  ") WITH (" 
                    +  "   'connector'  =  'values' ," 
                    +  "   'bounded'  =  ' true ' " 
                    +  ")" ;
    util().tableEnv().executeSql(ddl);
    util().verifyRelPlan( "WITH MView AS (SELECT CASE\n" 
            +  "  WHEN a = b THEN a\n" 
            +  "  ELSE CAST(NULL AS STRING)\n" 
            +  "  END AS a\n" 
            +  "  FROM MTable)\n" 
            +  "SELECT a FROM MView WHERE a IS NOT NULL" );
}
