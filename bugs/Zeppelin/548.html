        if (null == binder) {
          logger.info("2.10 binder is null");
        }
      } else {
        logger.info("scala is 2.11");
        binder = (Map<String, Object>) getLastObject();
        if (null == binder) {
          logger.info("2.11 binder is null");
        }
      }
      if (null != binder) {
        logger.info("binder is not null");
