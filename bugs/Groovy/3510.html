Title

Description

 import  groovy.transform.MapConstructor

@CompileStatic  //  static  bug only
  class  GroovyMapConstructorCheck {

    @MapConstructor(noArg =  true )
     class  Goo {
         final   int  x0

             return   "Goo(|$x0|)"
