Title

Non-deterministic functions return different values even if it is referred with the same column name

Description

@Test
 def  testRand(): Unit = {
  checkResult(
    s"""
       |SELECT b - a FROM (
       |  SELECT r + 5 AS a, r + 7 AS b FROM (
       |    SELECT RAND() AS r FROM SmallTable3
       |  ) t1
       |) t2
       |""".stripMargin,
    Seq(row(2), row(2), row(2))
  )
}
