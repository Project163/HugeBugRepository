```js
const fastify = require('fastify')({ logger: true })
fastify.get('/:strangeId', (req, reply) => {
  reply.send({ strangeId: req.params.strangeId })
})
fastify.listen(3000)
```

```

-H 'Connection: keep-alive'   -H 'Host: localhost:3000'   -H 'User-Agent: PostmanRuntime/7.17.1'
```
