Title

Dropping varchar partition throw exception

Description
Drop partition fail if the partition column is varchar. For example: 
  
 
create external table BS_TAB_0_211494(c_date_SAD_29630 date) PARTITIONED BY (part_varchar_37229 varchar(56)) STORED AS orc;

INSERT INTO BS_TAB_0_211494 values( '4740-04-04' , 'BrNTRsv3c' );

ALTER TABLE BS_TAB_0_211494 DROP PARTITION (part_varchar_37229= 'BrNTRsv3c' );
