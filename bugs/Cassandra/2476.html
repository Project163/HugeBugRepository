Title

Description

                ...     name text PRIMARY KEY, 
                ...     owner text, 
                ...     date timestamp, 
                ...     tags set<text> 
                ... ); 

 cqlsh:userkeyspace> delete tags ['cuddly']  from images where name = 'cat.jpg'; // not allowd cql statement

cqlsh:userkeyspace> describe table plays;
