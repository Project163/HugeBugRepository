Title

Description

spark.sql( "set spark.sql.codegen.fallback= false " )

val df = Seq(( "100-200" ,  "1" ,  "300" )).toDF( "a" ,  "b" ,  "c" )
val df2 = df.select(regexp_replace($ "a" ,  "(\\d+)" ,  "num" ))
df2.mapPartitions(x => x)(RowEncoder(df2.schema)).collect()
