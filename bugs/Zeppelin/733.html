 public class LongRuntime {
  private long value;

  public LongRuntime(long runtime) {
    this.value = runtime;
  }

  public long getValue() {
    return value;
  }
}
 
  

  
 public class NotebookImportLongDeserializer implements JsonDeserializer<LongRuntime> {

  @Override
  public LongRuntime deserialize(JsonElement jsonElement, Type typeOF,
    JsonDeserializationContext context) throws JsonParseException {
    try {
      if (jsonElement.isJsonObject()) {
        JsonObject obj = jsonElement.getAsJsonObject();
        return new LongRuntime(obj.get("value").getAsLong());
      } else {
        String elem = jsonElement.getAsString();
        if (elem.contains(".")) {
          return new LongRuntime(new Double(jsonElement.getAsDouble()).longValue());
        } else {
          return new LongRuntime(jsonElement.getAsLong());
        }
      }
    } catch (NumberFormatException e) {
      throw new NumberFormatException("Unparsable number: \"" + jsonElement.getAsString());
    }
  }
}
