Title

Description

 It uses this java class: 

  
  package  playground.groovy;

 import  java.util.List;

 public   class  Foo {

     private  List< String > names;

     public  List< String > getNames() {
         return  names;
    }

     public  void setNames(List< String > names) {
         this .names = names;
    }
}
