Title

Description

 CREATE   DATABASE  db2;
 CREATE   TABLE  db2.destinTable (one  STRING , two  STRING ) PARTITIONED  BY  (ds  STRING );
 INSERT  OVERWRITE  TABLE  db2.destinTable  PARTITION  (ds= '2011-11-11' )  IF   NOT   EXISTS   SELECT  100, 200;
