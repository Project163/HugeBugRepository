Title

Description

    val tableEnv = TableEnvironment.getTableEnvironment(env, config)
    val in = testData(env).toTable(tableEnv).as( 'a, ' b, 'c)
    val func0 =  new  TableFunc0

    val result = in
      .join(func0('c))
      .where(StartWith( 'name,  "J" ) && (' age > 20))

      .toDataSet[Row]

    val results = result.collect()
