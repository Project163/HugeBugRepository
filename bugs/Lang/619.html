Title

SerializationUtils.clone(Object) throws ClassCastException when called with a Serializable lambda

Description
SerializationUtils does not clone serializable lambda. 
  
 
 import  java.io.Serializable;
 import  java.util.function.Supplier;

 import  org.junit.jupiter.api.Test;
 import   static  org.junit.jupiter.api.Assertions.assertEquals;

 public   class  SerializationUtilsTest {

     interface  SerializableSupplier<T>  extends  Supplier<T>, Serializable {

    }

    @Test
    void testCloneLambda() {
        SerializableSupplier< String > supplier = () ->  "test" ;
        assertEquals( "test" , supplier.get());

         // next lines works fine
         SerializableSupplier< String > springClone = org.springframework.util.SerializationUtils.clone(supplier);
        assertEquals( "test" , springClone.get());

         // line below produces an error
         SerializableSupplier< String > commonsClone = org.apache.commons.lang3.SerializationUtils.clone(supplier);
        assertEquals( "test" , commonsClone.get());
    }

}
