- To see the difference edit  go/src/pkg/net/http/transport.go  and add a log line as follows: 

  
 func (es *bodyEOFSignal) Read(p [] byte ) (n  int , err error) {
	es.mu.Lock()
	closed, rerr := es.closed, es.rerr
	es.mu.Unlock()
	 if  closed {
		 return  0, errors.New( "http: read on closed response body" )
	}
	 if  rerr != nil {
		 return  0, rerr
	}

	n, err = es.body.Read(p)
	 // log.Printf( "%T: n=%d err=%q" , es, n, err)
 	 if  err != nil {
		es.mu.Lock()
		defer es.mu.Unlock()
		 if  es.rerr == nil {
			es.rerr = err
		}
		es.condfn(err)
	}
	 return 
}
