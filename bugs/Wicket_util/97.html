Title

Jupiter engine transitively included in war file

Description
If I run mvn dependency:tree on this basic war pom: 
  
 <?xml version="1.0"?>
<project xmlns="[URL] xmlns:xsi="[URL] xsi:schemaLocation="[URL] [URL]
	<modelVersion>4.0.0</modelVersion>
	<groupId>test</groupId>
	<artifactId>test</artifactId>
	<packaging>war</packaging>
	<version>1.0</version>
	<dependencies>
			<dependency>
				<groupId>org.apache.wicket</groupId>
				<artifactId>wicket-core</artifactId>
				<version>9.0.0</version>
				<type>jar</type>
				<scope>compile</scope>
			</dependency>
	</dependencies>
</project>
