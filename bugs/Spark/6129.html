        sql(
          """
            |CREATE TABLE `tab1` (`col1` INT, `TDATE` DATE)

          """.stripMargin)
        spark.table( "tab1" ).where( "TDATE >=  '2017-08-15' " ).createOrReplaceTempView( "tmpView1" )
