Title

Remove dummy new line at the end of non-sql commands

Description
In Driver.java 

  
 ss = Utilities.readColumn(resStream, bos);

  row = new String(bos.getData(), 0, bos.getCount(), "UTF-8");

  row = new String();
}

if (row != null) {
  numRows++;
  res.add(row);
}
