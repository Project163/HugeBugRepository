Title

Description

       <artifactId>maven-surefire-plugin</artifactId> 
       <configuration> 
         <environmentVariables> 
           <key>value</key> 
         </environmentVariables> 
         <forkMode>once</forkMode> 
       </configuration> 
     </plugin>
