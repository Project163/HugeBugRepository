Title

channelInactive() not getting called when ctx.close() is called on the ChannelHandlerContext

Description
### Expected behavior
channelInactive() gets called when ctx.close() is called on the ChannelHandlerContext

### Actual behavior
channelInactive() does not get called when ctx.close() is called on the ChannelHandlerContext.

### Steps to reproduce
We have a test where we write invalid xml to a test channel. ctx.close() is called within exceptionCaught() and I would expect channelInactive to fire. The test is as below. Assert.assertTrue(listener.connectionClosed) is currently failing whereas it was previously passing when using netty-handler 4.0.28.Final. 

        listener.connectionError = null;
        Assert.assertFalse(listener.connectionClosed);
        TestUtils.writeInbound(channel, xml5);
        Assert.assertEquals(null, readNext());
        Assert.assertNotNull(listener.connectionError);
        Assert.assertTrue(listener.connectionClosed);

### Minimal yet complete reproducer code (or URL to code)

    @Override
    public void exceptionCaught(ChannelHandlerContext ctx, Throwable exc) {
        if (connectionListener == null) {
            return;
        }
        // Treat unexpected exceptions as fatal to the connection
        try {
            connectionListener.connectionError(exc);
        } finally {
            ctx.close();
        }
    }

    @Override
    public void channelInactive(ChannelHandlerContext ctx) throws Exception {
        if (connectionListener == null) {
            return;
        }
        connectionListener.connectionClosed();
    }

### Netty version
netty-all 4.1.11.FINAL

class XmppEventDispatcher extends ChannelInboundHandlerAdapter {

    private static final Logger LOG = MchLoggerFactory.getLogger(XmppEventDispatcher.class);

    private final ConnectionListener connectionListener;

    /**
     * @param connectionListener Listener to dispatch events to. If null, no events will be dispatched.
     */
    public XmppEventDispatcher(ConnectionListener connectionListener) {
        this.connectionListener = connectionListener;
    }

    @Override
    public void channelRead(ChannelHandlerContext ctx, Object msg) {
        if (connectionListener == null) {
            return;
        }

        if (msg instanceof StreamError) {
            StreamErrorException exc = new StreamErrorException((StreamError)msg);
            connectionListener.connectionError(exc);
            // As per XMPP spec, server will close connection after a stream error
        } else {
            connectionListener.accept(msg);
        }
    }

    @Override
    public void channelInactive(ChannelHandlerContext ctx) throws Exception {
        if (connectionListener == null) {
            return;
        }
        connectionListener.connectionClosed();
    }

    @Override
    public void exceptionCaught(ChannelHandlerContext ctx, Throwable exc) {
        if (exc instanceof DecoderException) {
            // Log and ignore messages with XMPP extensions that we don't support
            Throwable cause = exc.getCause();
            if (cause instanceof XmppMarshalException) {
                LOG.error("Ignoring unparseable message: {}", cause.getMessage(), cause);
                return;
            }
        }

        if (connectionListener == null) {
            return;
        }

        // Treat unexpected exceptions as fatal to the connection
        try {
            connectionListener.connectionError(exc);
        } finally {
            ctx.close();
        }
    }
}
