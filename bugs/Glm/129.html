Title

Bug in glm::pow for quaternions

Description
In glm/gtx/quaternion.inl, the function pow (reproduced below) provides incorrect results. For example, glm::pow(glm::quat(0,1,0,0), 1.0f) returns the quaternion (1, nan, nan, nan). A fix is provided below
#### original function

template  
    GLM_FUNC_QUALIFIER tquat  pow
    (
        tquat  const & x,
        T const & y
    )
    {
        if(abs(x.w) > (static_cast (1) - epsilon ()))
            return x;
        T Angle = acos(y);
        T NewAngle = Angle \* y;
        T Div = sin(NewAngle) / sin(Angle);
        return tquat (
            cos(NewAngle),
            x.x \* Div,
            x.y \* Div,
            x.z \* Div);
    }
#### fixed function

template  
    GLM_FUNC_QUALIFIER tquat  pow
    (
        tquat  const & x,
        T const & y
    )
    {
            if(std::abs(x.w) > (static_cast (1) - glm::epsilon ())) {
            return x;
        }
        T Angle = acos(x.w);
        T NewAngle = Angle \* y;
        T Div = sin(NewAngle) / sqrt(x.x_x.x + x.y_x.y + x.z*x.z);
        return glm::tquat (
                            cos(NewAngle),
                            x.x \* Div,
                            x.y \* Div,
                            x.z \* Div);
    }
