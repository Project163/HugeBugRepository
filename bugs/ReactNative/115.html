Title

Destructuring object with rest operator failed when using Number keys

Description
```
const obj = { 1: 'v1', 2: 'v2' };
const key = 1;

const { [String(key)]: a, ...b } = obj;
console.log(a, b); // a is 'v1', b is { 2: 'v2' }

const { [key]: c, ...d } = obj;
console.log(c, d); // c is 'v1', d is { 1: 'v1', 2: 'v2' }
```
