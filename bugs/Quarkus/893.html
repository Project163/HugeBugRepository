1. Add the below package type(quarkus.package.type=mutable-jar)
2. oc new-build registry.access.redhat.com/ubi8/openjdk-11:1.3 --binary --name=quarkus-remote -l app=quarkus-remote
3. rm -rf target/binary && mkdir -p target/binary && cp -r target/remote-dev-demo-1.0.0-SNAPSHOT-runner.jar target/lib target/binary
