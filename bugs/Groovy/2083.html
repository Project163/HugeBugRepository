Title

Description

   String  name

   static  Iterator<Element> single() {
    [  new  Element( name:  'single'  ) ].iterator()
  }

   static  Iterator<List<Element>> multi() {
    [ [  new  Element( name:  'first'  ),  new  Element( name:  'second'  ) ] ].iterator()
