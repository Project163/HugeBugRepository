Title

Constructor chaining in enums doesn't work

Description

  enum  Foo {
  ONE(1), TWO(1, 2)
  
   int  i
   int  j
  
  Foo( int  i) {
     this (i, 0)  //  if  we set i and j manually here, all is fine
   }
  
  Foo( int  i,  int  j) {
     this .i = i
     this .j = j
  }
}

def foos = [Foo.ONE, Foo.TWO]
