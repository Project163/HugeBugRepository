- i had same issue **"Could not debug. The "path" argument must be of type string. Received type undefined"**

resolved by removing             
"outDir": "${workspaceRoot}/.vscode/.react"
and added
            "cwd": "${workspaceRoot}"
in launch.json

code sample:

{
            "name": "Debug Android",
            "program": "${workspaceRoot}/.vscode/launchReactNative.js",
            "type": "reactnative",
            "request": "launch",
            "platform": "android",
            "sourceMaps": true,
            "cwd": "${workspaceRoot}"
        },
        {
            "name": "Debug iOS",
            "program": "${workspaceRoot}/.vscode/launchReactNative.js",
            "type": "reactnative",
            "request": "launch",
            "platform": "ios",
            "sourceMaps": true,
            "cwd": "${workspaceRoot}"
        },
