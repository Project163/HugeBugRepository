Title

Description

 interface  SomeResource {
    @PUT
    @Consumes(MediaType.MULTIPART_FORM_DATA)
    void uploadFile(@Multipart( "bin" )  byte [] fileBytes, 
                             @Multipart(value =  "timestamp" , type =  "text/plain" )  long  timestamp);
}
