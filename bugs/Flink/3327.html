Title

Description
sql : 

 val rank = 
 """ 
 
  
 
 SELECT 
 
 
  gradeId, 
 
 
  classId, 
 
 
  stuId, 
 
 
  score, 
 
 
  dense_rank() OVER (PARTITION BY gradeId, classId ORDER BY score asc) as dense_rank_num 
 
 
 FROM student 
 
 
 
  
 

  """.stripMargin

 row("grade2", "class2", "0006", 90),
