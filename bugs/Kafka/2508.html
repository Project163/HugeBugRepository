- If anyone runs into this and needs a workaround before getting 2.4.1/trunk. Might need some tweaking based on where/how you log and obviously different if you aren't using systemd, will need user/dirs tweaked to be a general workaround 
  
 
journalctl -u kafka --since  'yesterday'  | grep  'Resetting first dirty offset of'  | awk  '{print $14 ,$19}'  | sort -u > /home/user/checkpoints
cp /kafka-topic-data/cleaner-offset-checkpoint /home/user
cat /home/user/checkpoints | sed -r  's/(.*)-/\1 /g'  > /home/user/checkpoints-processed
cat /home/user/checkpoints-processed cleaner-offset-checkpoint | sort --key=1,2 -u > /home/user/cleaner-offset-checkpoint.clean
sudo systemctl stop kafka; sudo mv /kafka-topic-data/cleaner-offset-checkpoint /home/user/; sudo mv /home/user/cleaner-offset-checkpoint.clean /kafka-topic-data/cleaner-offset-checkpoint; sudo chown -R kafka:kafka /kafka-topic-data/cleaner-offset-checkpoint; sleep 5; sudo systemctl start kafka
