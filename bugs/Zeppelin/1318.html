<dependency> <groupId>io.grpc</groupId> <artifactId>grpc-stub</artifactId> <version>${grpc.version}</version> <exclusions> <exclusion> <groupId>com.google.errorprone</groupId> <artifactId>error_prone_annotations</artifactId> </exclusion> </exclusions> </dependency>

 ./mvnw -pl rlang -am -Phadoop3  -Pbuild-distr   package  -DskipTests
