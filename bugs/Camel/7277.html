Description

      - unmarshal:
          beanio:
            mapping:  "mappings.xml" 
            streamName:  "personStream"

      - split:
          simple:  "${body}" 
          steps:
            - to:  "mock:beanio-unmarshal"

<beanio xmlns= "[URL]

     </stream> 

 </beanio>
