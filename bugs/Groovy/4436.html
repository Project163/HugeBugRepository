Title

Type variable inference does not work properly in nested diamond operators

Description
I have the following program 

  
 
 class  A<T> {
  T f;
  A(T f) {
     this .f = f;
  }
}

 class  B<T> {
  A<T> f;
  B(A<T> f) {
     this .f = f;
  }
}

 class  Test {
  void test() {
    def x =  new   B<>( new  A<>(( long ) 1)).f;
    A< Long > y = x;
  }
  
}
