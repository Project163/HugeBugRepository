- Sounds like you made some customizations to next.config.js to add support for image imports before Next.js 11. I've opened a PR to mention what to do when next.config.js was customized to support this (disable the feature or migrate to the built-in support). You can check out the docs update in #27104.

Thanks for the feedback!
- @timneutkens actually I don't see any config options related to importing images in my next's.config.js. Here it is:

```javascript
const { parsed: localEnv } = require('dotenv').config();
const webpack = require('webpack');
const { withSentryConfig } = require('@sentry/nextjs');
const withTM = require('next-transpile-modules')(['next-auth', 'yup']);
const path = require('path');
const withBundleAnalyzer = require('@next/bundle-analyzer')({
	enabled: process.env.ANALYZE === 'true',
});
const { i18n } = require('./next-i18next.config');
const rewrites = require('./constants/rewrites');

module.exports = withBundleAnalyzer(
	withSentryConfig(
		withTM({
			defaultLocale: 'ru-Ru',
			poweredByHeader: false,
			images: {
				domains: [' ', 'localhost'],
			},
			async rewrites() {
				return rewrites;
			},
			eslint: {
				// Warning: Dangerously allow production builds to successfully complete even if
				// your project has ESLint errors.
				ignoreDuringBuilds: true,
			},
			reactStrictMode: true,
			i18n,
			webpack(config) {
				config.plugins.push(new webpack.EnvironmentPlugin(localEnv));
				// Here is the magic
				// We push our config into the resolve.modules array
				config.resolve.modules.push(path.resolve('./'));

				return config;
			},
		}),
	),
);
`
