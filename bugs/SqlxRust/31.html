async fn sqlx_time() -> anyhow::Result  {
    let mut opts = sqlx::postgres::PgConnectOptions::new();

        .username("postgres")
        .password("zalando")
        .database("citus")
        .port(19000);

    let mut client = sqlx::postgres::PgConnection::connect_with(&opts).await?;
    let now = std::time::Instant::now();

    let _: Vec  = sqlx::query(Q).fetch_all(&mut client).await?;

    Ok(now.elapsed())
}
